```python
"""Fetch and cache OSRS wiki content. Throwaway audit tool, not committed."""

import hashlib
import json
import pathlib
import time
import urllib.parse
import urllib.request

WIKI = "https://oldschool.runescape.wiki"
API = f"{WIKI}/api.php"
UA = "reinze-db-audit/1.0 (claude@ai.ryanohara.co)"
CACHE_DIR = pathlib.Path(__file__).parent / "cache"
DELAY_SECONDS = 0.1


def cache_key(title: str) -> str:
    """Filesystem-safe, stable, collision-resistant name for a page title."""
    return hashlib.sha256(title.encode("utf-8")).hexdigest()


def _http_get(url: str) -> str:
    request = urllib.request.Request(url, headers={"User-Agent": UA})
    with urllib.request.urlopen(request, timeout=30) as response:
        body = response.read().decode("utf-8")
    time.sleep(DELAY_SECONDS)
    return body


def fetch_raw(title: str) -> str:
    """Page wikitext, cached on disk so re-runs cost no network calls."""
    CACHE_DIR.mkdir(parents=True, exist_ok=True)
    path = CACHE_DIR / f"{cache_key(title)}.wikitext"
    if path.exists():
        return path.read_text(encoding="utf-8")
    url = f"{WIKI}/w/{urllib.parse.quote(title)}?action=raw"
    body = _http_get(url)
    path.write_text(body, encoding="utf-8")
    return body


def list_transclusions(template: str) -> list[str]:
    """Mainspace page titles transcluding `template`, following continuation."""
    titles: list[str] = []
    cont = None
    while True:
        params = {
            "action": "query",
            "format": "json",
            "list": "embeddedin",
            "eititle": template,
            "einamespace": "0",
            "eilimit": "500",
        }
        if cont:
            params["eicontinue"] = cont
        data = json.loads(_http_get(f"{API}?{urllib.parse.urlencode(params)}"))
        titles.extend(p["title"] for p in data["query"]["embeddedin"])
        cont = data.get("continue", {}).get("eicontinue")
        if not cont:
            return titles
```

```python
"""Parse OSRS wiki infobox templates. Throwaway audit tool, not committed."""

import re


def extract_template(wikitext: str, name: str) -> dict[str, str] | None:
    """Fields of the named template, or None if the template is not present.

    Scans brace depth from the template's opening `{{` so nested templates in
    field values (`[[File:x.png]]`, `{{Coins|...}}`) do not end the block early.
    """
    match = re.search(r"\{\{\s*" + re.escape(name) + r"\s*[\|\n}]", wikitext)
    if not match:
        return None

    start = match.start()
    depth = 0
    end = None
    i = start
    while i < len(wikitext) - 1:
        pair = wikitext[i:i + 2]
        if pair == "{{":
            depth += 1
            i += 2
            continue
        if pair == "}}":
            depth -= 1
            i += 2
            if depth == 0:
                end = i
                break
            continue
        i += 1
    if end is None:
        return None

    body = wikitext[start + 2:end - 2]

    # Split on pipes at depth zero only, so `{{Coins|1|2}}` stays one value.
    parts: list[str] = []
    depth = 0
    current = ""
    i = 0
    while i < len(body):
        pair = body[i:i + 2]
        if pair in ("{{", "[["):
            depth += 1
            current += pair
            i += 2
            continue
        if pair in ("}}", "]]"):
            depth -= 1
            current += pair
            i += 2
            continue
        if body[i] == "|" and depth == 0:
            parts.append(current)
            current = ""
            i += 1
            continue
        current += body[i]
        i += 1
    parts.append(current)

    fields: dict[str, str] = {}
    for part in parts[1:]:  # parts[0] is the template name
        if "=" not in part:
            continue
        key, value = part.split("=", 1)
        fields[key.strip().lower()] = value.strip()
    return fields


def parse_number(raw: str) -> float | None:
    """`"2,230"` -> 2230.0. Returns None for anything not a plain number."""
    cleaned = raw.replace(",", "").strip()
    if not re.fullmatch(r"\d+(\.\d+)?", cleaned):
        return None
    return float(cleaned)


def _key(name: str) -> str:
    return name.strip().replace(" ", "_")


def construction_entries(wikitext: str) -> list[tuple[str, float]]:
    """(key, xp) pairs for one Infobox Construction page.

    One entry under the base name when every version shares an experience
    value; one entry per distinct value keyed `Name_(Version)` when they differ.
    """
    fields = extract_template(wikitext, "Infobox Construction")
    if not fields:
        return []
    name = fields.get("name", "").strip()
    if not name:
        return []

    flat = parse_number(fields.get("experience", ""))
    if flat is not None:
        return [(_key(name), flat)]

    versioned: list[tuple[str, float]] = []
    index = 1
    while f"experience{index}" in fields:
        xp = parse_number(fields[f"experience{index}"])
        version = fields.get(f"version{index}", "").strip()
        if xp is not None:
            versioned.append((version, xp))
        index += 1

    if not versioned:
        return []

    distinct = {xp for _, xp in versioned}
    if len(distinct) == 1:
        return [(_key(name), versioned[0][1])]

    out: list[tuple[str, float]] = []
    seen: set[float] = set()
    for version, xp in versioned:
        if xp in seen:
            continue
        seen.add(xp)
        label = f"{name} ({version})" if version else name
        out.append((_key(label), xp))
    return out


def prayer_entry(wikitext: str) -> tuple[str, float, str] | None:
    """(key, xp, type) for one Prayer info page, or None."""
    fields = extract_template(wikitext, "Prayer info")
    if not fields:
        return None
    name = fields.get("name", "").strip()
    xp = parse_number(fields.get("xp", ""))
    if not name or xp is None:
        return None
    return (_key(name), xp, fields.get("type", "").strip())
```

```python
"""Compare INI sections against wiki-derived data. Throwaway, not committed."""


def read_section(ini_text: str, section: str) -> dict[str, str]:
    """key -> raw value for one INI section, in file order."""
    out: dict[str, str] = {}
    inside = False
    for line in ini_text.split("\n"):
        stripped = line.strip()
        if stripped.startswith("[") and stripped.endswith("]"):
            inside = stripped == f"[{section}]"
            continue
        if not inside or not stripped or "=" not in stripped:
            continue
        key, value = stripped.split("=", 1)
        out[key] = value
    return out


def format_number(xp: float) -> str:
    """The single place a number becomes INI text."""
    return str(int(xp)) if xp == int(xp) else str(xp)


def reconcile(ini: dict[str, str], wiki: dict[str, float]) -> dict[str, list]:
    """Sort into unchanged / changed / wiki_only / ini_only.

    Keys are matched case-insensitively because the lookup in params.rs is.
    """
    ini_by_lower = {k.lower(): k for k in ini}
    wiki_by_lower = {k.lower(): k for k in wiki}

    unchanged: list[str] = []
    changed: list[tuple[str, str, str]] = []
    ini_only: list[tuple[str, str]] = []
    wiki_only: list[tuple[str, str]] = []

    for lower, ini_key in ini_by_lower.items():
        if lower not in wiki_by_lower:
            ini_only.append((ini_key, ini[ini_key]))
            continue
        want = format_number(wiki[wiki_by_lower[lower]])
        have = ini[ini_key].strip()
        if have == want:
            unchanged.append(ini_key)
        else:
            changed.append((ini_key, ini[ini_key], want))

    for lower, wiki_key in wiki_by_lower.items():
        if lower not in ini_by_lower:
            wiki_only.append((wiki_key, format_number(wiki[wiki_key])))

    return {
        "unchanged": unchanged,
        "changed": changed,
        "wiki_only": sorted(wiki_only),
        "ini_only": ini_only,
    }
```

```python
"""Produce the reconcile reports a human works from. Throwaway, not committed."""

import pathlib
import sys

sys.path.insert(0, str(pathlib.Path(__file__).parent))

import fetch
import parse
import reconcile

INI_PATH = pathlib.Path(
    "/home/rohara/.agent-deck/multi-repo-worktrees/788b004e/rust-reinze/lib/Database.ini"
)
HERE = pathlib.Path(__file__).parent


def gather_construction() -> tuple[dict[str, float], list[str]]:
    titles = fetch.list_transclusions("Template:Infobox Construction")
    data: dict[str, float] = {}
    skipped: list[str] = []
    for i, title in enumerate(titles, 1):
        if i % 50 == 0:
            print(f"  construction {i}/{len(titles)}", file=sys.stderr)
        entries = parse.construction_entries(fetch.fetch_raw(title))
        if not entries:
            skipped.append(title)
        for key, xp in entries:
            data[key] = xp
    return data, skipped


def gather_prayer() -> tuple[dict[str, float], list[str]]:
    titles = fetch.list_transclusions("Template:Prayer info")
    data: dict[str, float] = {}
    skipped: list[str] = []
    for i, title in enumerate(titles, 1):
        if i % 50 == 0:
            print(f"  prayer {i}/{len(titles)}", file=sys.stderr)
        entry = parse.prayer_entry(fetch.fetch_raw(title))
        if entry is None:
            skipped.append(title)
            continue
        key, xp, kind = entry
        data[key] = xp
    return data, skipped


def write_report(path: pathlib.Path, section: str, buckets: dict, skipped: list[str]):
    lines = [f"=== {section} reconcile report ===", ""]
    lines.append(f"unchanged: {len(buckets['unchanged'])} entries (not listed)")
    lines.append("")

    lines.append(f"--- CHANGED ({len(buckets['changed'])}) : ini -> wiki")
    for key, have, want in sorted(buckets["changed"]):
        lines.append(f"  {key}: {have} -> {want}")
    lines.append("")

    lines.append(f"--- WIKI-ONLY / candidate additions ({len(buckets['wiki_only'])})")
    for key, value in buckets["wiki_only"]:
        lines.append(f"  {key}={value}")
    lines.append("")

    lines.append(
        f"--- INI-ONLY / decide each by hand ({len(buckets['ini_only'])})"
    )
    for key, value in sorted(buckets["ini_only"]):
        lines.append(f"  {key}={value}")
    lines.append("")

    lines.append(f"--- PAGES WITH NO PARSEABLE TEMPLATE ({len(skipped)})")
    for title in skipped:
        lines.append(f"  {title}")
    lines.append("")

    path.write_text("\n".join(lines), encoding="utf-8")
    print(f"wrote {path}")


def main():
    import json

    ini_text = INI_PATH.read_text(encoding="utf-8")

    print("gathering Prayer...", file=sys.stderr)
    prayer, prayer_skipped = gather_prayer()
    write_report(
        HERE / "report-prayer.txt",
        "Prayer",
        reconcile.reconcile(reconcile.read_section(ini_text, "Prayer"), prayer),
        prayer_skipped,
    )
    (HERE / "wiki-prayer-keys.json").write_text(
        json.dumps(sorted(prayer), indent=1), encoding="utf-8"
    )

    print("gathering Construction...", file=sys.stderr)
    construction, construction_skipped = gather_construction()
    write_report(
        HERE / "report-construction.txt",
        "Construction",
        reconcile.reconcile(
            reconcile.read_section(ini_text, "Construction"), construction
        ),
        construction_skipped,
    )
    (HERE / "wiki-construction-keys.json").write_text(
        json.dumps(sorted(construction), indent=1), encoding="utf-8"
    )


if __name__ == "__main__":
    main()
```
